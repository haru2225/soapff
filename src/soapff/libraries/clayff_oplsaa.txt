# soapff library: ClayFF (Cygan, Liang, Kalinichev 2004) and OPLS-AA (oplsaa) as one force field, for minerals with water,
# ions and organic molecules, as Szczerba & Kalinichev (Clays Clay Miner. 64, 488 (2016)) put the two together: the LJ of
# unlike atoms by the Lorentz-Berthelot rules (arithmetic sigma, geometric epsilon) for every pair, the 1-4 pairs of
# OPLS-AA scaled by 0.5 (ClayFF has none).
#
# The OPLS-AA types, parameters and references are those of oplsaa (extends); this file adds the ClayFF types (Table 1),
# the O-H bonds and the H-O-H and M-O-H angles (Table 2) with the flexible SPC water, as in clayff.txt, and the references
# of clayff.  The metals and bridging O of ClayFF are named in no row of [bonds]: their bonds of step 1 only give the SOAP
# its neighbourhood, and the force field has no metal-O bond.

[library]
name        = clayff_oplsaa
extends     = oplsaa
description = ClayFF minerals, water and ions with OPLS-AA molecules in one force field (Lorentz-Berthelot for every pair)
reference   = R. T. Cygan, J.-J. Liang, A. G. Kalinichev, J. Phys. Chem. B 108, 1255 (2004); W. L. Jorgensen, D. S. Maxwell, J. Tirado-Rives, J. Am. Chem. Soc. 118, 11225 (1996); M. Szczerba, A. G. Kalinichev, Clays Clay Miner. 64, 488 (2016)

[bond_lengths]
# E1 E2 r(A), as in clayff.txt (metal-O halfway between the longest bond and the shortest distance not bonded in the
# references); the C pairs and H-O come from oplsaa.txt
# E1 E2  r       d_b    d_n
O  Si  2.41   # 1.683  3.143
Al O   2.66   # 2.105  3.213
Mg O   2.70   # 2.187  3.217

[lammps]
# ClayFF's mixing for every pair; the other settings are those of oplsaa (special_bonds lj/coul 0.0 0.0 0.5, ...)
pair_style     = lj/cut/coul/long 10.0
pair_modify    = mix arithmetic tail no
timestep       = 0.5
