# soapff library: montmorillonite with edges (T. R. Underwood, I. C. Bourg, J. Phys. Chem. C 124, 3702 (2020), after Lammers et
# al.), the optional clay library of soapff.
#
# engine clayff with the force field of the reference particle taken whole from its data and settings files
# (core.read_forcefield: charges, LJ, O-H bond, M-O-H and edge metal-O-metal angles of every type); the water and the ions
# added are ClayFF.  Typed in the two steps of clayff.txt, the water molecules taken apart from the particle (a water O
# nearer than the bond length to an edge Al or Mg is not bonded to it).  Montmorillonite only: other minerals stop.

[library]
name        = clayff_edge
engine      = clayff
description = montmorillonite particle with edges (Underwood & Bourg 2020); added water and ions ClayFF
reference   = T. R. Underwood, I. C. Bourg, J. Phys. Chem. C 124, 3702 (2020)
forcefield  = clayff/edge/clay_montmorillonite.data clayff/edge/clay_montmorillonite.settings
forcefield_name = Montmorillonite edge
forcefield_source = T. R. Underwood, I. C. Bourg, J. Phys. Chem. C 124, 3702 (2020)
prebuilt    = clayff/prebuilt/library_clayff_edge.npz
prebuilt_sha = 52e1cee2e8e5251b
prebuilt_dscribe = 2.1.2

[soap]
r_cut = 5.0
n_max = 4
l_max = 3
sigma = 0.4
depth = 4
threshold = 0.9

[bond_lengths]
# E1 E2 r(A): two atoms of these elements nearer than r are bonded (step 1).  r lies between the longest bond (d_b)
# and the shortest distance of two atoms not bonded (d_n) in the references (checked when the table is built); the
# metal-O pairs halfway.  H-O by hand, as in clayff.txt.
# E1 E2  r       d_b    d_n
O  Si  2.41   # 1.647  3.183
Al O   2.65   # 1.963  3.333
Mg O   2.65   # 1.963  3.333
H  O   1.25   # 1.000  1.806  (halfway 1.40)
