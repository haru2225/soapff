# soapff library: the solution library of clayff (its second step: the water and the ions).  Not used alone.
#
# The 22 mineral structures of clayff and the structures that packmol builds, not relaxed (no MD frame): liquid water (512
# molecules), the NaCl, KCl, CsCl, CaCl2 and BaCl2 solutions, and the same water and solutions in vacuum (films, a droplet).
# The water is in the SOAP; the rows of the systems with water are pruned at 0.99, the slab rows at 0.999.
# (examples/library_water/build_library_initial.py)

[library]
name        = clayff_water
engine      = clayff
hidden      = yes
description = second step of clayff: water (flexible SPC) and the ions of ClayFF Table 1
reference   = R. T. Cygan, J.-J. Liang, A. G. Kalinichev, J. Phys. Chem. B 108, 1255 (2004)
prebuilt    = clayff/prebuilt/library_clayff_water.npz
prebuilt_sha = f9d9ea7853fb5dcd

[soap]
r_cut = 5.0
n_max = 4
l_max = 3
sigma = 0.4
water_in_soap = yes
threshold = 0.9
